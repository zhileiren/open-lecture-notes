\begin{frame}[t,fragile]{os}
\begin{verbatim}
>>> import os
>>> os.system("date")
2020年 11月 08日 星期日 17:07:33 CST
>>> os.path.exists("a.txt")
True
>>> os.getcwd()
'/home/ren'
>>> os.chdir("/tmp")
>>> os.mkdir("/tmp/abc")
>>> if os.path.exists("a.txt"):
...     os.rename("a.txt", "b.txt")

>>> os.remove("b.txt")

\end{verbatim}

	
\end{frame}

\begin{frame}[t,fragile]{subprocess}
\begin{verbatim}

>>> import subprocess
>>> subprocess.call(['dir', '/a'])
>>> subprocess.call("dir /a", shell=True)
>>> return_data = subprocess.call(['ls'], stdout=subprocess.PIPE)
>>> result_txt = return_data.stdout.decode("utf-8")
\end{verbatim}
	
\end{frame}

\begin{frame}[t,fragile]{glob}

>>> for f in glob("**/*.txt", recursive=True):
...     print(f)
a.txt
b.txt
c.txt
d/a.txt
...
\end{verbatim}
	
\end{frame}
