\begin{frame}[t,fragile]{发起连接}
\begin{verbatim}
import requests
r = requests.get("http://127.0.0.1:8000")
print(r.text)
print(r.encoding)
print(r.url)
print(r.status_code)
\end{verbatim}
    
\end{frame}

\begin{frame}[t,fragile]{GET参数}
\begin{verbatim}
import requests
import json
payload = {'key1': 'value1', 'key2': 'value2'}
r = requests.get('http://httpbin.org/get', params=payload)
result = json.loads(r.text)
print(result['args'])

\end{verbatim}
    
\end{frame}
