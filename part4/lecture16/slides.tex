\begin{frame}[t]{约束求解的历史}

    \begin{enumerate}
        \item SAT: The satisfiability problem
        \item SMT: Satisfiability modulo theories
    \end{enumerate}
    
\end{frame}

\begin{frame}[t]{约束求解示例}
Consider the following puzzle. Spend exactly 100 dollars and buy exactly 100 animals. Dogs cost 15 dollars, cats cost 1 dollar, and mice cost 25 cents each. You have to buy at least one of each. How many of each should you buy?     
\end{frame}

\begin{frame}[t,fragile]{约束求解示例}
    \begin{verbatim}
import z3
dog, cat, mouse = z3.Ints('dog cat mouse')
z3.solve(dog >= 1,   # at least one dog
      cat >= 1,      # at least one cat
      mouse >= 1,    # at least one mouse
      # we want to buy 100 animals
      dog + cat + mouse == 100,
      # We have 100 dollars (10000 cents):
      #   dogs cost 15 dollars (1500 cents), 
      #   cats cost 1 dollar (100 cents), and 
      #   mice cost 25 cents 
      1500 * dog + 100 * cat + 25 * mouse == 10000)
\end{verbatim}
    
\end{frame}

\begin{frame}[t,fragile]{程序生成/修复}
\begin{verbatim}
from z3 import *
x = Int('x')
y = Int('y')
f = Function('f', IntSort(), IntSort())
s = Solver()
s.add(f(f(x)) == x, f(x) == y, x != y)
print s.check()
m = s.model()
print("f(f(x)) =", m.evaluate(f(f(x))))
print("f(x)    =", m.evaluate(f(x)))
\end{verbatim}
\end{frame}
